package main

import (
	"context"
	"encoding/json"
	"fmt"
	"io/fs"
	"os"

	"cercano/source/server/internal/anthropicauth"

	"cercano/source/server/internal/hostsvc/credentials"
	"cercano/source/server/internal/secrets"
	"cercano/source/server/pkg/accountidentity"
)

type backing map[string]string

var _ secrets.Store = backing{}

func (m backing) Get(k string) (string, error) {
	v, ok := m[k]
	if !ok {
		return "", fs.ErrNotExist
	}
	return v, nil
}
func (m backing) Set(k, v string) error { m[k] = v; return nil }
func (m backing) Delete(k string) error { delete(m, k); return nil }
func (m backing) List() ([]string, error) {
	keys := []string{}
	for k := range m {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	return keys, nil
}
func must(err error) {
	if err != nil {
		panic(err)
	}
}
func equal(label string, a, b any) {
	if !reflect.DeepEqual(a, b) {
		panic("assertion failed: " + label)
	}
}
func identity(tag string) accountidentity.Identity {
	return accountidentity.Identity{Email: tag + "@example.invalid", Name: "Synthetic " + tag + " Ω"}
}
func anth(tag string) anthropicauth.TokenSet {
	return anthropicauth.TokenSet{Identity: identity(tag), Access: "synthetic-access-" + tag, Refresh: "synthetic-refresh-" + tag, ExpiresAt: time.Unix(1700000123, 456000000).UTC()}
}
func chat(tag string) chatgptauth.TokenSet {
	return chatgptauth.TokenSet{Identity: identity(tag), Access: "synthetic-access-" + tag, Refresh: "synthetic-refresh-" + tag, ExpiresAt: time.Unix(1800000123, 789000000).UTC(), AccountID: "synthetic-account-" + tag}
}
func encoded(provider, tag string) string {
	var raw string
	var err error
	if provider == "anthropic" {
		raw, err = anth(tag).Encode()
	} else {
		raw, err = chat(tag).Encode()
	}
	must(err)
	return raw
}
func commit(s *credentials.Service, profile, provider, tag string) {
	a, err := s.BeginLogin(context.Background(), profile, provider)
	must(err)
	defer a.Close()
	must(a.Commit(encoded(provider, tag)))
}
func seedExpected(pop bool) backing {
	m := backing{}
	if pop {
		m["sentinel"] = "synthetic-untouched-Ω\x00opaque"
		m["opaque/work"] = "synthetic-old"
		m["delete/me"] = "synthetic-delete"
		for _, p := range []string{"anthropic", "openai-responses"} {
			m[p+"/work"] = encoded(p, p+"-old")
			m[p+"/other"] = encoded(p, p+"-other")
		}
	}
	return m
}
func finalExpected(pop bool) backing {
	m := seedExpected(pop)
	m["opaque/work"] = "synthetic-replaced"
	m["opaque/new"] = "synthetic-new\x00Ω"
	delete(m, "delete/me")
	for _, p := range []string{"anthropic", "openai-responses"} {
		m[p+"/work"] = encoded(p, p+"-new")
		m[p+"/added"] = encoded(p, p+"-added")
	}
	return m
}
func verify(s *credentials.Service, m, want backing, pop, final bool) {
	equal("entire backing map / exact changes", m, want)
	keys, err := s.List()
	must(err)
	sort.Strings(keys)
	expected, _ := want.List()
	equal("profile list", keys, expected)
	for k, v := range want {
		got, err := s.Get(k)
		must(err)
		equal("opaque bytes "+k, got, v)
	}
	for _, p := range []string{"anthropic", "openai-responses"} {
		tags := map[string]string{}
		if pop {
			tags["work"] = p + "-old"
			tags["other"] = p + "-other"
		}
		if final {
			tags["work"] = p + "-new"
			tags["added"] = p + "-added"
		}
		for profile, tag := range tags {
			raw, err := s.Get(p + "/" + profile)
			must(err)
			if p == "anthropic" {
				got, err := anthropicauth.DecodeTokenSet(raw)
				must(err)
				equal("all Anthropic fields "+profile, got, anth(tag))
			} else {
				got, err := chatgptauth.DecodeTokenSet(raw)
				must(err)
				equal("all ChatGPT fields including AccountID "+profile, got, chat(tag))
			}
		}
	}
	if final {
		if _, err := s.Get("delete/me"); err == nil {
			panic("deleted credential still readable")
		}
	}
}
func main() {
	if len(os.Args) != 4 {
		panic("mode path empty|populated required")
	}
	mode, path, kind := os.Args[1], os.Args[2], os.Args[3]
	if kind != "empty" && kind != "populated" {
		panic("unknown scenario")
	}
	pop := kind == "populated"
	m := backing{}
	if mode != "seed" {
		b, err := os.ReadFile(path)
		must(err)
		must(json.Unmarshal(b, &m))
	}
	s := credentials.New(m)
	switch mode {
	case "seed":
		for k, v := range seedExpected(pop) {
			must(s.Set(k, v))
		}
		verify(s, m, seedExpected(pop), pop, false)
	case "seed-check":
		verify(s, m, seedExpected(pop), pop, false)
	case "write":
		verify(s, m, seedExpected(pop), pop, false)
		// Set then replace even in the empty scenario; deletion is exercised there too.
		must(s.Set("opaque/work", "synthetic-intermediate"))
		must(s.Set("opaque/work", "synthetic-replaced"))
		must(s.Set("opaque/new", "synthetic-new\x00Ω"))
		if !pop {
			must(s.Set("delete/me", "synthetic-delete"))
		}
		must(s.Delete("delete/me"))
		for _, p := range []string{"anthropic", "openai-responses"} {
			commit(s, p+"/work", p, p+"-new")
			commit(s, p+"/added", p, p+"-added")
		}
		verify(s, m, finalExpected(pop), pop, true)
	case "verify":
		verify(s, m, finalExpected(pop), pop, true)
	default:
		panic("unknown mode")
	}
	if mode == "seed" || mode == "write" {
		b, err := json.Marshal(m)
		must(err)
		must(os.WriteFile(path, b, 0600))
	}
	fmt.Println("PASS", kind, mode, "exact map, profiles, opaque bytes and all applicable token fields")
}
